# Build options for RexGraph.

# pip install . -Csetup-args=-Dnative=true

# 'native' enables machine-specific optimization (-march=native -ffast-math).
# It is OFF by default so the default build is portable across CPUs and keeps
# IEEE-correct floating point. Turn it on only for a build that will run on the
# same machine (or an identical CPU) that compiled it.
option(
    'native',
    type: 'boolean',
    value: false,
    description: 'Optimize for the build CPU (-march=native -ffast-math). Not portable; relaxes IEEE fp.',
)

# 'blas' controls what happens when no BLAS/LAPACK is found at configure time.
#
# The compiled extensions call cblas_* and LAPACK from _linalg.pxd. Without a BLAS
# to link against they still COMPILE, and then fail at import with
# "undefined symbol: cblas_dgemm". That is how a broken wheel gets published: the
# build warns, the warning scrolls past in a few hundred lines of compiler output,
# and the artifact looks fine until someone installs it somewhere that does not
# already have a BLAS loaded.
#
# So the default is 'auto', which now ERRORS rather than warning. Use 'disabled'
# only if you know the symbols will be present at runtime some other way:
#   pip install . -Csetup-args=-Dblas=disabled
option(
    'blas',
    type: 'feature',
    value: 'auto',
    description: 'Require a BLAS/LAPACK at build time. disabled = build anyway, symbols must come from elsewhere at runtime.',
)

# 'blas_provider' names the BLAS/LAPACK to link. 'auto' searches for OpenBLAS and then
# for a separate BLAS and LAPACK. 'accelerate' links Apple's Accelerate framework, which
# exports the unprefixed cblas_* and LAPACK symbols the kernels call and puts nothing
# into a wheel. 'scipy-openblas' links the library of the scipy-openblas32 wheel from
# 'blas_dir'; that build prefixes every symbol with scipy_, so each symbol the kernels
# call is renamed at compile time.
option(
    'blas_provider',
    type: 'combo',
    choices: ['auto', 'accelerate', 'scipy-openblas'],
    value: 'auto',
    description: 'BLAS/LAPACK to link: auto, accelerate (macOS), or scipy-openblas from blas_dir.',
)

option('blas_dir', type: 'string', value: '',
       description: 'directory of the BLAS library for blas_provider=scipy-openblas')

# 'openmp' controls the OpenMP lanes. 'auto' uses OpenMP where the compiler provides it.
# A macOS wheel disables it: Homebrew's libomp is built for the runner's own macOS
# release, and bundling it would raise the wheel's minimum macOS to that release.
option('openmp', type: 'feature', value: 'auto',
       description: 'Build the OpenMP lanes where the compiler provides OpenMP.')

# GPU architecture for the optional HIP ternary lane, e.g. gfx1151. Empty asks rocminfo.
option('hip_arch', type: 'string', value: '',
       description: 'offload arch for the HIP ternary kernel (empty: detect)')
