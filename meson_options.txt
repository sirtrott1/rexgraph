# Build options for RexGraph.

# pip install . -Csetup-args=-Dnative=true

# 'native' enables machine-specific optimization (-march=native -ffast-math).
# It is OFF by default so the default build is portable across CPUs and keeps
# IEEE-correct floating point. Turn it on only for a build that will run on the
# same machine (or an identical CPU) that compiled it.
option(
    'native',
    type: 'boolean',
    value: false,
    description: 'Optimize for the build CPU (-march=native -ffast-math). Not portable; relaxes IEEE fp.',
)

# 'blas' controls what happens when no BLAS/LAPACK is found at configure time.
#
# The compiled extensions call cblas_* and LAPACK from _linalg.pxd. Without a BLAS
# to link against they still COMPILE, and then fail at import with
# "undefined symbol: cblas_dgemm". That is how a broken wheel gets published: the
# build warns, the warning scrolls past in a few hundred lines of compiler output,
# and the artifact looks fine until someone installs it somewhere that does not
# already have a BLAS loaded.
#
# So the default is 'auto', which now ERRORS rather than warning. Use 'disabled'
# only if you know the symbols will be present at runtime some other way:
#   pip install . -Csetup-args=-Dblas=disabled
option(
    'blas',
    type: 'feature',
    value: 'auto',
    description: 'Require a BLAS/LAPACK at build time. disabled = build anyway, symbols must come from elsewhere at runtime.',
)
